# OrderBook Component

Displays all open buy and sell orders with one-click order filling functionality.

## 📋 Overview

**File:** `src/components/OrderBook.js`

**Purpose:**
- Display all open buy orders (bids)
- Display all open sell orders (asks)
- Enable one-click order filling
- Show real-time order updates
- Provide price discovery

## 🏗️ Component Structure

```javascript
import React, { Component } from 'react'
import { connect } from 'react-redux'
import { OverlayTrigger, Tooltip } from 'react-bootstrap'
import Spinner from './Spinner'
import {
  orderBookSelector,
  orderBookLoadedSelector,
  exchangeSelector,
  accountSelector,
  orderFillingSelector
} from '../store/selectors'
import { fillOrder } from '../store/interactions'

class OrderBook extends Component {
  render() {
    return (
      <div className="vertical">
        <div className="card bg-dark text-white">
          <div className="card-header">Order Book</div>
          <div className="card-body order-book">
            {this.props.showOrderBook ? (
              <div>
                {this.renderBuyOrders()}
                <hr />
                {this.renderSellOrders()}
              </div>
            ) : (
              <Spinner type="table" />
            )}
          </div>
        </div>
      </div>
    )
  }

  renderBuyOrders() {
    const { orderBook } = this.props
    
    return (
      <div>
        <table className="table table-dark table-sm small">
          <thead>
            <tr>
              <th>DAPP</th>
              <th>DAPP/ETH</th>
              <th>ETH</th>
            </tr>
          </thead>
          <tbody>
            {orderBook.buyOrders.map((order) => this.renderOrder(order, 'buy'))}
          </tbody>
        </table>
      </div>
    )
  }

  renderSellOrders() {
    const { orderBook } = this.props
    
    return (
      <div>
        <table className="table table-dark table-sm small">
          <thead>
            <tr>
              <th>DAPP</th>
              <th>DAPP/ETH</th>
              <th>ETH</th>
            </tr>
          </thead>
          <tbody>
            {orderBook.sellOrders.map((order) => this.renderOrder(order, 'sell'))}
          </tbody>
        </table>
      </div>
    )
  }

  renderOrder(order, type) {
    const { dispatch, exchange, account } = this.props
    
    return (
      <OverlayTrigger
        key={order.id}
        placement='auto'
        overlay={
          <Tooltip id={order.id}>
            {`Click here to ${type === 'buy' ? 'sell' : 'buy'}`}
          </Tooltip>
        }
      >
        <tr
          className={`order-book-order order-${type}`}
          onClick={(e) => fillOrder(dispatch, exchange, order, account)}
        >
          <td>{order.tokenAmount}</td>
          <td className={`text-${type === 'buy' ? 'success' : 'danger'}`}>
            {order.tokenPrice}
          </td>
          <td>{order.etherAmount}</td>
        </tr>
      </OverlayTrigger>
    )
  }
}

function mapStateToProps(state) {
  const orderBookLoaded = orderBookLoadedSelector(state)
  const orderFilling = orderFillingSelector(state)

  return {
    orderBook: orderBookSelector(state),
    showOrderBook: orderBookLoaded && !orderFilling,
    exchange: exchangeSelector(state),
    account: accountSelector(state)
  }
}

export default connect(mapStateToProps)(OrderBook)
```

## 🔄 Data Processing

### Order Decoration

**Add calculated fields:**
```javascript
const decorateOrderBookOrders = (orders) => {
  return orders.map((order) => {
    order = decorateOrder(order)
    order = decorateOrderBookOrder(order)
    return order
  })
}

const decorateOrder = (order) => {
  let etherAmount
  let tokenAmount

  // Determine order type
  if (order.tokenGive === ETHER_ADDRESS) {
    etherAmount = order.amountGive
    tokenAmount = order.amountGet
  } else {
    etherAmount = order.amountGet
    tokenAmount = order.amountGive
  }

  // Calculate the token price to 5 decimal places
  const precision = 100000
  let tokenPrice = (etherAmount / tokenAmount)
  tokenPrice = Math.round(tokenPrice * precision) / precision

  return {
    ...order,
    etherAmount: ether(etherAmount),
    tokenAmount: tokens(tokenAmount),
    tokenPrice,
    formattedTimestamp: moment.unix(order.timestamp).format('h:mm:ss a M/D')
  }
}

const decorateOrderBookOrder = (order) => {
  const orderType = order.tokenGive === ETHER_ADDRESS ? 'buy' : 'sell'

  return {
    ...order,
    orderType,
    orderTypeClass: (orderType === 'buy' ? 'success' : 'danger'),
    orderFillAction: (orderType === 'buy' ? 'sell' : 'buy')
  }
}
```

## 🎯 Advanced Features

### 1. Order Grouping

**Group by price level:**
```javascript
groupOrdersByPrice(orders) {
  const grouped = {}
  
  orders.forEach(order => {
    const price = order.tokenPrice.toFixed(5)
    
    if (!grouped[price]) {
      grouped[price] = {
        price: price,
        tokenAmount: 0,
        etherAmount: 0,
        orders: []
      }
    }
    
    grouped[price].tokenAmount += parseFloat(order.tokenAmount)
    grouped[price].etherAmount += parseFloat(order.etherAmount)
    grouped[price].orders.push(order)
  })
  
  return Object.values(grouped)
}

renderGroupedOrders() {
  const grouped = this.groupOrdersByPrice(this.props.orderBook.buyOrders)

      <td>{level.etherAmount.toFixed(4)}</td>

}
```

### 2. Depth Chart

**Visual representation of liquidity:**
```javascript
renderDepthChart() {
  const buyDepth = this.calculateDepth(this.props.orderBook.buyOrders)
  const sellDepth = this.calculateDepth(this.props.orderBook.sellOrders)
  
  return (
    <div className="depth-chart">
      {/* Render bars for each price level */}
      {buyDepth.map(level => (
        <div 
          key={level.price}
          className="depth-bar buy"
          style={{ width: `${level.percentage}%` }}
        />
      ))}
    </div>
  )
}

calculateDepth(orders) {
  let cumulative = 0
  const maxVolume = orders.reduce((sum, o) => 
    sum + parseFloat(o.etherAmount), 0
  )
  
  return orders.map(order => {
    cumulative += parseFloat(order.etherAmount)
    return {
      price: order.tokenPrice,
      volume: cumulative,
      percentage: (cumulative / maxVolume) * 100
    }
  })
}
```

### 3. Spread Display

**Show bid-ask spread:**
```javascript
renderSpread() {
  const { buyOrders, sellOrders } = this.props.orderBook
  
  if (buyOrders.length === 0 || sellOrders.length === 0) {
    return null
  }
  
  const bestBid = buyOrders[0].tokenPrice
  const bestAsk = sellOrders[0].tokenPrice
  const spread = bestAsk - bestBid
  const spreadPercent = (spread / bestAsk) * 100
  
  return (
    <div className="spread-display">
      <small className="text-muted">
        Spread: {spread.toFixed(5)} ETH ({spreadPercent.toFixed(2)}%)
      </small>
    </div>
  )
}
```

### Fill Order Action

```javascript
// store/interactions.js
export const fillOrder = (dispatch, exchange, order, account) => {
  dispatch(orderFilling())
  
  exchange.methods.fillOrder(order.id)
    .send({ from: account })
    .on('transactionHash', (hash) => {
      console.log('Transaction hash:', hash)
    })
    .on('error', (error) => {
      console.error(error)
      window.alert('There was an error filling the order!')
    })
}
```

## 🐛 Common Issues

### Issue 1: Orders Not Displaying

### Issue 2: Fill Order Fails

**Problem:** Transaction reverts when filling order

**Solution:** Validate before filling
```javascript
handleFillOrder = async (order) => {
  const { account, exchange, token } = this.props
  
  // Check if order is still open
  const orderFilled = await exchange.methods.orderFilled(order.id).call()
  if (orderFilled) {
    window.alert('Order already filled!')
    return
  }
  
  // Check user has enough balance
  const requiredBalance = order.orderType === 'buy' 
    ? order.tokenAmount 
    : order.etherAmount
    
  const balance = await this.getBalance(account)
  if (balance < requiredBalance) {
    window.alert('Insufficient balance!')
    return
  }
  
  // Proceed with fill
  fillOrder(dispatch, exchange, order, account)
}
```

### Issue 3: Incorrect Sorting

**Problem:** Orders not sorted by best price

**Solution:** Verify sort logic
```javascript
// Buy orders: highest price first (descending)
const buyOrders = orders
  .filter(o => o.orderType === 'buy')
  .sort((a, b) => {

const sellOrders = orders
```
