# PriceChart Component

Interactive candlestick chart displaying token price movements using ApexCharts.

## 📋 Overview

**File:** `src/components/PriceChart.js`

**Purpose:**
- Visualize price movements over time
- Display OHLC (Open, High, Low, Close) data
- Show trading volume
- Provide interactive chart controls
- Real-time price updates

## 🏗️ Component Structure

```javascript
import React, { Component } from 'react'
import { connect } from 'react-redux'
import Chart from 'react-apexcharts'
import Spinner from './Spinner'
import {
  priceChartLoadedSelector,
  priceChartSelector
} from '../store/selectors'

const priceSymbol = (lastPriceChange) => {
  let output
  if(lastPriceChange === '+') {
    output = <span className="text-success">&#9650;</span> // Green up
  } else {
    output = <span className="text-danger">&#9660;</span> // Red down
  }
  return(output)
}

class PriceChart extends Component {
  render() {
    return (
      <div className="card bg-dark text-white">
        <div className="card-header">Price Chart</div>
        <div className="card-body">
          {this.props.priceChartLoaded ? (
            this.renderChart()
          ) : (
            <Spinner type="div" />
          )}
        </div>
      </div>
    )
  }

  renderChart() {
    return (
      <div className="price-chart">
        <div className="price-stats">
          <span className="price-label">DAPP/ETH {priceSymbol(this.props.priceChart.lastPriceChange)}</span>
          <span className="price-value">{this.props.priceChart.lastPrice}</span>
        </div>
        <Chart
          options={this.props.priceChart.options}
          series={this.props.priceChart.series}
          type='candlestick'
          width='100%'
          height='100%'
        />
      </div>
    )
  }
}

function mapStateToProps(state) {
  return {
    priceChartLoaded: priceChartLoadedSelector(state),
    priceChart: priceChartSelector(state)
  }
}

export default connect(mapStateToProps)(PriceChart)
```

## 🔄 Data Processing

### Building Chart Data

```javascript
// store/selectors.js
export const priceChartSelector = createSelector(
  filledOrders,
  orders => {
    // Sort by timestamp
    orders = orders.sort((a, b) => a.timestamp - b.timestamp)
    
    // Format data for chart
    orders = orders.map(o => decorateOrder(o))
    
    // Group by hour
    let groupedOrders = groupBy(orders, (o) => 
      moment.unix(o.timestamp).startOf('hour').format()
    )
    
    // Build candlestick data
    const series = Object.keys(groupedOrders).map(hour => {
      const group = groupedOrders[hour]
      
      return {
        x: new Date(hour),
        y: [
          group[0].tokenPrice,                    // Open
          Math.max(...group.map(o => o.tokenPrice)), // High
          Math.min(...group.map(o => o.tokenPrice)), // Low
          group[group.length - 1].tokenPrice      // Close
        ]
      }
    })
    
    // Calculate last price change
    const lastPriceChange = series.length >= 2
      ? series[series.length - 1].y[3] >= series[series.length - 2].y[3] ? '+' : '-'
      : '+'
    
    return {
      series: [{ name: 'Price', data: series }],
      options: chartOptions,
      lastPrice: series.length > 0 
        ? series[series.length - 1].y[3].toFixed(5) 
        : '0.00000',
      lastPriceChange
    }
  }
)
```

### Time Grouping

**Group by different intervals:**
```javascript
groupOrdersByTime(orders, interval = 'hour') {
  return groupBy(orders, (order) => {
    switch(interval) {
      case 'minute':
        return moment.unix(order.timestamp).startOf('minute').format()
      case 'hour':
        return moment.unix(order.timestamp).startOf('hour').format()
      case 'day':
        return moment.unix(order.timestamp).startOf('day').format()
      default:
        return moment.unix(order.timestamp).startOf('hour').format()
    }
  })
}
```

### OHLC Calculation

```javascript
calculateOHLC(trades) {
  if (trades.length === 0) {
    return { open: 0, high: 0, low: 0, close: 0 }
  }
  
  const prices = trades.map(t => parseFloat(t.tokenPrice))
  
  return {
    open: prices[0],
    high: Math.max(...prices),
    low: Math.min(...prices),
    close: prices[prices.length - 1]
  }
}
```

```javascript
getFilteredData() {
  const { timeRange } = this.state
  const now = moment()
  let cutoff
  
  switch(timeRange) {
    case '5m':
      cutoff = now.subtract(5, 'minutes')
      break
    case '15m':
      cutoff = now.subtract(15, 'minutes')
      break
    case '1h':
      cutoff = now.subtract(1, 'hour')
      break
    case '4h':
      cutoff = now.subtract(4, 'hours')
      break
    case '1d':
      cutoff = now.subtract(1, 'day')
      break
    case '1w':
      cutoff = now.subtract(1, 'week')
      break
    default:
      cutoff = now.subtract(1, 'hour')
  }
  
  return this.props.priceChart.series[0].data.filter(point =>
    moment(point.x).isAfter(cutoff)
  )
}
```

```javascript
calculateVolume() {
  // Group trades and sum volumes
  const groupedTrades = groupBy(
    this.props.filledOrders,
    o => moment.unix(o.timestamp).startOf('hour').format()
  )
  
  return Object.keys(groupedTrades).map(time => ({
    x: new Date(time),
    y: groupedTrades[time].reduce((sum, trade) => 
      sum + parseFloat(trade.etherAmount), 0
    )
  }))
}
```

```javascript
calculateMovingAverage(data, period) {
  const ma = []
  
  for (let i = period - 1; i < data.length; i++) {
    const slice = data.slice(i - period + 1, i + 1)
    const sum = slice.reduce((acc, point) => 
      acc + point.y[3], 0 // Use close price
    )
    ma.push({
      x: data[i].x,
      y: sum / period
    })
  }
  
  return ma
}
```

**RSI (Relative Strength Index):**
```javascript
calculateRSI(data, period = 14) {
  const changes = []
  
  for (let i = 1; i < data.length; i++) {
    changes.push(data[i].y[3] - data[i-1].y[3])
  }
  
  const gains = changes.map(c => c > 0 ? c : 0)
  const losses = changes.map(c => c < 0 ? Math.abs(c) : 0)
  
  const avgGain = gains.slice(0, period).reduce((a, b) => a + b) / period
  const avgLoss = losses.slice(0, period).reduce((a, b) => a + b) / period
  
  const rs = avgGain / avgLoss
  const rsi = 100 - (100 / (1 + rs))
  
  return rsi
}

renderRSI() {
  const rsi = this.calculateRSI(this.props.priceChart.series[0].data)
  
  return (
    <div className="rsi-indicator">
      <span className="label">RSI(14):</span>
      <span className={`value ${
        rsi > 70 ? 'text-danger' : 
        rsi < 30 ? 'text-success' : 
        'text-muted'
      }`}>
        {rsi.toFixed(2)}
      </span>
    </div>
  )
}
```

### 5. Price Alerts

```javascript
state = {
  priceAlerts: []
}

addPriceAlert(price, direction) {
  this.setState(prev => ({
    priceAlerts: [
      ...prev.priceAlerts,
      {
        id: Date.now(),
        price: parseFloat(price),
        direction, // 'above' or 'below'
        triggered: false
      }
    ]
  }))
}

componentDidUpdate(prevProps) {
  if (prevProps.priceChart.lastPrice !== this.props.priceChart.lastPrice) {
    this.checkPriceAlerts()
  }
}

checkPriceAlerts() {
  const currentPrice = parseFloat(this.props.priceChart.lastPrice)
  
  this.state.priceAlerts.forEach(alert => {
    if (!alert.triggered) {
      const shouldTrigger = 
        (alert.direction === 'above' && currentPrice >= alert.price) ||
        (alert.direction === 'below' && currentPrice <= alert.price)
      
      if (shouldTrigger) {
        this.triggerAlert(alert)
      }
    }
  })
}

triggerAlert(alert) {
  // Show notification
  new Notification('Price Alert', {
    body: `Price ${alert.direction} ${alert.price}`,
    icon: '/logo.png'
  })
  
  // Mark as triggered
  this.setState(prev => ({
    priceAlerts: prev.priceAlerts.map(a =>
      a.id === alert.id ? { ...a, triggered: true } : a
    )
  }))
}
```

## 🔄 Real-time Updates

### WebSocket Integration

```javascript
componentDidMount() {
  this.subscribeToTrades()
}

subscribeToTrades() {
  const { exchange } = this.props
  
  this.tradeSubscription = exchange.events.Trade({})
    .on('data', (event) => {
      this.handleNewTrade(event)
    })
}

handleNewTrade(event) {
  // Update chart with new trade
  const trade = {
    timestamp: event.returnValues.timestamp,
    price: this.calculatePrice(event.returnValues)
  }
  
  this.updateChart(trade)
}

componentWillUnmount() {
  if (this.tradeSubscription) {
    this.tradeSubscription.unsubscribe()
  }
}
```

### Issue 3: Performance with Large Datasets

**Problem:** Chart lags with many data points

**Solution:** Implement data decimation
```javascript
decimateData(data, maxPoints = 100) {
  if (data.length <= maxPoints) return data
  
  const step = Math.ceil(data.length / maxPoints)
  return data.filter((_, index) => index % step === 0)
}
```
