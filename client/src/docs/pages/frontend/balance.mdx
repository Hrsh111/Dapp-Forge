# Balance Component

Displays user balances in both wallet and exchange, with deposit and withdrawal functionality.

## 📋 Overview

**File:** `src/components/Balance.js`

**Purpose:**
- Display ETH and token balances
- Show wallet balances (in MetaMask)
- Show exchange balances (deposited funds)
- Provide deposit functionality
- Provide withdrawal functionality

## 🏗️ Component Structure

```javascript
import React, { Component } from 'react'
import { connect } from 'react-redux'
import { Tabs, Tab } from 'react-bootstrap'
import Spinner from './Spinner'
import {
  loadBalances,
  depositEther,
  depositToken,
  withdrawEther,
  withdrawToken
} from '../store/interactions'
import {
  web3Selector,
  accountSelector,
  tokenSelector,
  exchangeSelector,
  etherBalanceSelector,
  tokenBalanceSelector,
  exchangeEtherBalanceSelector,
  exchangeTokenBalanceSelector,
  balancesLoadingSelector
} from '../store/selectors'

class Balance extends Component {
  state = {
    etherDepositAmount: 0,
    tokenDepositAmount: 0,
    etherWithdrawAmount: 0,
    tokenWithdrawAmount: 0
  }

  render() {
    return (
      <div className="card bg-dark text-white">
        <div className="card-header">Balance</div>
        <div className="card-body">
          {this.props.showForm ? (
            this.renderForm()
          ) : (
            <Spinner />
          )}
        </div>
      </div>
    )
  }

  renderForm() {
    const {
      etherBalance,
      tokenBalance,
      exchangeEtherBalance,
      exchangeTokenBalance
    } = this.props

    return (
      <Tabs defaultActiveKey="deposit" className="bg-dark text-white">
        <Tab eventKey="deposit" title="Deposit" className="bg-dark">
          {this.renderDepositForm()}
        </Tab>
        <Tab eventKey="withdraw" title="Withdraw" className="bg-dark">
          {this.renderWithdrawForm()}
        </Tab>
      </Tabs>
    )
  }
}

function mapStateToProps(state) {
  const balancesLoading = balancesLoadingSelector(state)

  return {
    web3: web3Selector(state),
    account: accountSelector(state),
    token: tokenSelector(state),
    exchange: exchangeSelector(state),
    etherBalance: etherBalanceSelector(state),
    tokenBalance: tokenBalanceSelector(state),
    exchangeEtherBalance: exchangeEtherBalanceSelector(state),
    exchangeTokenBalance: exchangeTokenBalanceSelector(state),
    balancesLoading,
    showForm: !balancesLoading
  }
}

export default connect(mapStateToProps)(Balance)
```

### Balance Update Flow

```
1. User clicks Deposit/Withdraw
       ↓
2. Form submission handler called
       ↓
3. Convert amount to Wei
       ↓
4. Call interaction function
       ↓
5. Send transaction to blockchain
       ↓
6. Smart contract emits event
       ↓
7. Event listener catches update
       ↓
8. Redux state updated
       ↓
9. Component re-renders with new balances
```

```javascript
export const depositToken = async (
  dispatch,
  exchange,
  web3,
  token,
  amount,
  account
) => {
  try {
    dispatch(balancesLoading())
    
    // Step 1: Approve tokens
    await token.methods.approve(exchange.options.address, amount)
      .send({ from: account })
      .on('transactionHash', (hash) => {
        // Approval pending
      })
    
    // Step 2: Deposit tokens
    await exchange.methods.depositToken(token.options.address, amount)
      .send({ from: account })
      .on('transactionHash', (hash) => {
        dispatch(tokenDepositMade())
      })
      .on('error', (error) => {
        console.error(error)
        window.alert('There was an error!')
      })
  } catch (error) {
    console.error(error)
  }
}
```

### Balance Validation

```javascript
validateDeposit(amount, availableBalance) {
  const amountWei = this.props.web3.utils.toWei(amount, 'ether')
  
  if (parseFloat(amount) <= 0) {
    window.alert('Amount must be greater than 0')
    return false
  }
  
  if (parseFloat(amountWei) > parseFloat(availableBalance)) {
    window.alert('Insufficient balance')
    return false
  }
  
  return true
}

handleDepositEther = (event) => {
```

## 🐛 Common Issues

### Issue 1: Balance Not Updating

**Problem:** Balances don't update after transaction

**Solution:** Subscribe to events
```javascript
// In App component
exchange.events.Deposit({}, (error, event) => {
  if (!error) {
    loadBalances(dispatch, web3, exchange, token, account)
  }
})
```

### Issue 2: Token Approval Fails

**Problem:** Token deposit fails at approval step

**Solution:** Check allowance first
```javascript
// Check current allowance
const currentAllowance = await token.methods
  .allowance(account, exchange.options.address)
  .call()

// If allowance sufficient, skip approval
if (parseInt(currentAllowance) >= parseInt(amount)) {
  // Proceed directly to deposit
} else {
  // Request approval first
}
```

### Issue 3: Insufficient Gas

**Problem:** Transactions fail due to gas

**Solution:** Estimate gas before sending
```javascript
try {
  const gasEstimate = await exchange.methods.depositEther()
    .estimateGas({ from: account, value: amount })
  
  await exchange.methods.depositEther()
    .send({ 
      from: account, 
      value: amount,
      gas: Math.floor(gasEstimate * 1.2) // 20% buffer
    })
} catch (error) {
  console.error('Gas estimation failed:', error)
}
```
