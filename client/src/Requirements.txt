# -------------------------------
# 1. System Prerequisites
# -------------------------------
# Install these system-level tools before running the project

Node.js >= 18.0.0
npm >= 9.0.0
git >= 2.30.0

# Recommended:
# Use Node Version Manager (nvm) for easy Node.js installation.


# -------------------------------
# 2. Global npm Packages
# -------------------------------
# Install globally to enable Truffle & Ganache across your system.
# Run:
#   npm install -g truffle ganache

truffle >= 5.11.5
ganache >= 7.9.0


# -------------------------------
# 3. Browser Extensions
# -------------------------------
# Required for interacting with the DApp via the browser

MetaMask (latest version)


# -------------------------------
# 4. Local Project Dependencies
# -------------------------------
# Automatically installed via:
#   npm install
#   cd client && npm install

# Backend / Blockchain
web3 >= 1.10.0
dotenv >= 16.3.1

# Frontend (React)
react >= 18.3.0
react-dom >= 18.3.0
bootstrap >= 5.3.2
@metamask/detect-provider >= 2.0.0

# Development utilities
eslint >= 8.56.0
prettier >= 3.2.0
